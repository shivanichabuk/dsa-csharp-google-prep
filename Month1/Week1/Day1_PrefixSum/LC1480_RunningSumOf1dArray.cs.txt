// LC1480 - Running Sum of 1d Array
// Pattern: Prefix Sum
// Time: O(n) | Space: O(1) in-place
// Approach: each element becomes sum of itself + all previous

public class Solution {
    public int[] RunningSum(int[] nums) {
        for (int i = 1; i < nums.Length; i++) {
            nums[i] += nums[i - 1];
        }
        return nums;
    }
}
